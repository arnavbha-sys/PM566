Question 2: Most frequent words

Removing stop words should make the results more informative.

Question 4: Bigrams and trigrams

Bigrams

The transcriptions were tokenized into bigrams, which are sequences of two consecutive words.

bigram_counts <- mt_samples |>
  filter(!is.na(transcription)) |>
  unnest_tokens(
    bigram,
    transcription,
    token = "ngrams",
    n = 2
  ) |>
  separate(
    bigram,
    into = c("word1", "word2"),
    sep = " ",
    remove = FALSE
  ) |>
  filter(
    !word1 %in% stop_words$word,
    !word2 %in% stop_words$word,
    !str_detect(word1, "[0-9]"),
    !str_detect(word2, "[0-9]")
  ) |>
  count(bigram, sort = TRUE)

Question 6: Most-used words in each specialty

specialty_word_counts <- mt_samples |>
  filter(
    !is.na(transcription),
    !is.na(medical_specialty)
  ) |>
  unnest_tokens(word, transcription) |>
  anti_join(stop_words, by = "word") |>
  filter(!str_detect(word, "[0-9]")) |>
  count(medical_specialty, word, sort = TRUE)
top_5_words_by_specialty <- specialty_word_counts |>
  group_by(medical_specialty) |>
  slice_max(
    order_by = n,

Question 7: Extra insight

To determine which words are especially characteristic of particular specialties, the proportion of each specialty’s words represented by each token was calculated.

specialty_word_proportions <- specialty_word_counts |>
  group_by(medical_specialty) |>
  mutate(
    total_words = sum(n),
    proportion = n / total_words
  ) |>
  ungroup()
distinctive_words_by_specialty <- specialty_word_proportions |>
  group_by(medical_specialty) |>
  slice_max(
    order_by = proportion,
    n = 5,
    with_ties = FALSE
  ) |>
  arrange(medical_specialty, desc(proportion)) |>
  ungroup()
